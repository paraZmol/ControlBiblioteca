    <script>
        // Logica de Tabs (Pestañas)
        function switchTab(tabName) {
            // Ocultar todos
            document.getElementById('tab-monitoreo').classList.add('hidden');
            document.getElementById('tab-historial').classList.add('hidden');
            
            // Resetear estilos de botones
            const inactiveClass = ['hover:bg-slate-100', 'dark:hover:bg-slate-800/50', 'text-slate-600', 'dark:text-slate-400', 'border-transparent'];
            const activeClass = ['bg-cyan-50', 'dark:bg-cyan-500/10', 'text-cyan-700', 'dark:text-cyan-400', 'border-cyan-200', 'dark:border-cyan-500/20'];
            
            document.getElementById('btn-monitoreo').classList.remove(...activeClass);
            document.getElementById('btn-monitoreo').classList.add(...inactiveClass);
            document.getElementById('btn-historial').classList.remove(...activeClass);
            document.getElementById('btn-historial').classList.add(...inactiveClass);

            // Activar el seleccionado
            document.getElementById(`tab-${tabName}`).classList.remove('hidden');
            document.getElementById(`btn-${tabName}`).classList.remove(...inactiveClass);
            document.getElementById(`btn-${tabName}`).classList.add(...activeClass);

            // Cambiar titulo
            document.getElementById('page-title').innerText = tabName === 'monitoreo' ? 'Monitoreo en Vivo' : 'Historial de Sesiones';
        }

        // Logica Dark Mode
        const themeToggleBtn = document.getElementById('theme-toggle');
        const themeIcon = document.getElementById('theme-icon');
        const htmlElement = document.documentElement;

        const setTheme = (isDark) => {
            if (isDark) {
                htmlElement.classList.add('dark');
                themeIcon.classList.remove('ph-moon');
                themeIcon.classList.add('ph-sun');
            } else {
                htmlElement.classList.remove('dark');
                themeIcon.classList.remove('ph-sun');
                themeIcon.classList.add('ph-moon');
            }
        };

        const isDarkMode = localStorage.theme === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches);
        setTheme(isDarkMode);

        themeToggleBtn.addEventListener('click', () => {
            const isDark = htmlElement.classList.toggle('dark');
            setTheme(isDark);
            localStorage.theme = isDark ? 'dark' : 'light';
        });
    </script>
</body>
